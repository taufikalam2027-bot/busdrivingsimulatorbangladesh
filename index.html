    public class SplashAndBootManager : MonoBehaviour
    {
        [Header("BS Team Branding UI")]
        [SerializeField] private CanvasGroup bsTeamLogoGroup;
        [SerializeField] private TextMeshProUGUI studioTitleText;
        [SerializeField] private float splashFadeDuration = 1.0f;
        [SerializeField] private float splashHoldDuration = 2.0f;

        [Header("Loading Screen UI")]
        [SerializeField] private GameObject loadingPanel;
        [SerializeField] private Slider progressBar;
        [SerializeField] private TextMeshProUGUI statusLabel;
        [SerializeField] private TextMeshProUGUI percentageLabel;

        [Header("Scene & Asset Configuration")]
        [SerializeField] private string garageSceneName = "Garage_Dhaka";
        [SerializeField] private string coreBangladeshiPackName = "bd_core_assets";
        [SerializeField] private string mapPackName = "bd_64districts_map";

        private void Start()
        {
            Application.targetFrameRate = 60;
            Screen.sleepTimeout = SleepTimeout.NeverSleep;

            if (studioTitleText != null)
                studioTitleText.text = "B.S TEAM";

            if (loadingPanel != null)
                loadingPanel.SetActive(false);

            StartCoroutine(ExecuteBootSequence());
        }

        private IEnumerator ExecuteBootSequence()
        {
            // Phase 1: Display B.S Team Splash identical to Maleo Team intro cadence
            yield return StartCoroutine(FadeCanvasGroup(bsTeamLogoGroup, 0f, 1f, splashFadeDuration));
            yield return new WaitForSeconds(splashHoldDuration);
            yield return StartCoroutine(FadeCanvasGroup(bsTeamLogoGroup, 1f, 0f, splashFadeDuration * 0.7f));

            if (loadingPanel != null)
                loadingPanel.SetActive(true);

            // Phase 2: Mount Bangladeshi Play Asset Delivery (PAD) or Legacy OBB
            UpdateProgress(0.05f, "Initializing BS Team Engine...");
            yield return StartCoroutine(VerifyAndMountBangladeshiAssets());

            // Phase 3: Asynchronously stream 3D Garage scene
            UpdateProgress(0.45f, "Loading Dhaka Garage & Hanif Fleet...");
            AsyncOperation loadOp = SceneManager.LoadSceneAsync(garageSceneName);
            loadOp.allowSceneActivation = false;

            while (loadOp.progress < 0.9f)
            {
                float mappedProgress = 0.45f + (loadOp.progress / 0.9f) * 0.50f;
                UpdateProgress(mappedProgress, "Preparing Bangladeshi Roads & Liveries...");
                yield return null;
            }

            UpdateProgress(1.0f, "Ready to Drive!");
            yield return new WaitForSeconds(0.35f);
            loadOp.allowSceneActivation = true;
        }

        private IEnumerator VerifyAndMountBangladeshiAssets()
        {
#if UNITY_ANDROID && !UNITY_EDITOR
            if (AndroidAssetPacks.coreUnityAssetPacksDownloaded)
            {
                string[] corePacks = AndroidAssetPacks.GetCoreUnityAssetPackNames();
                UpdateProgress(0.15f, $"Verified {corePacks.Length} Core Packs...");
            }

            // Verify both custom core fleet pack and 64-district map pack
            string[] targetPacks = new string[] { coreBangladeshiPackName, mapPackName };
            var stateOp = AndroidAssetPacks.GetAssetPackStateAsync(targetPacks);
            while (!stateOp.isDone)
                yield return null;

            bool needsDownload = false;
            foreach (var packState in stateOp.states)
            {
                if (packState.status != AndroidAssetPackStatus.Completed)
                {
                    needsDownload = true;
                    break;
                }
            }

            if (!needsDownload)
            {
                UpdateProgress(0.40f, "Bangladeshi Fleet & 64-District Map Mounted.");
            }
            else
            {
                UpdateProgress(0.25f, "Downloading Bangladesh Asset Packs...");
                var downloadOp = AndroidAssetPacks.DownloadAssetPackAsync(targetPacks);
                while (!downloadOp.isDone)
                {
                    UpdateProgress(0.25f + (downloadOp.progress * 0.15f), "Downloading Map & Traffic Assets...");
                    yield return null;
                }
            }
#else
            UpdateProgress(0.30f, "Verifying Local Bangladeshi Asset Bundles...");
            yield return new WaitForSeconds(0.5f);
#endif
        }

        private IEnumerator FadeCanvasGroup(CanvasGroup group, float startAlpha, float endAlpha, float duration)
        {
            if (group == null) yield break;
            float elapsed = 0f;
            group.alpha = startAlpha;

            while (elapsed < duration)
            {
                elapsed += Time.deltaTime;
                group.alpha = Mathf.Lerp(startAlpha, endAlpha, elapsed / duration);
                yield return null;
            }
            group.alpha = endAlpha;
        }

        private void UpdateProgress(float normalizedValue, string message)
        {
            if (progressBar != null)
                progressBar.value = Mathf.Clamp01(normalizedValue);
            if (statusLabel != null)
                statusLabel.text = message;
            if (percentageLabel != null)
                percentageLabel.text = $"{Mathf.RoundToInt(normalizedValue * 100f)}%";
        }
    }
}

    public class FloatingOriginManager : MonoBehaviour
    {
        public static FloatingOriginManager Instance { get; private set; }

        [Header("Target & Precision Threshold")]
        [SerializeField] private Transform playerBusTransform;
        [SerializeField] private float recenterThresholdMeters = 2000f;

        [Header("Tracked World Roots (Terrain, Roads, Landmarks, Traffic)")]
        [SerializeField] private List<Transform> worldRootObjects = new List<Transform>();

        public Vector3d CumulativeWorldOffset { get; private set; }

        [System.Serializable]
        public struct Vector3d
        {
            public double x, y, z;
            public Vector3d(double x, double y, double z) { this.x = x; this.y = y; this.z = z; }
        }

        private void Awake()
        {
            if (Instance != null && Instance != this)
            {
                Destroy(gameObject);
                return;
            }
            Instance = this;
            CumulativeWorldOffset = new Vector3d(0, 0, 0);
        }

        private void LateUpdate()
        {
            if (playerBusTransform == null) return;

            Vector3 currentPos = playerBusTransform.position;
            float horizontalDistance = new Vector2(currentPos.x, currentPos.z).magnitude;

            if (horizontalDistance >= recenterThresholdMeters)
            {
                ShiftWorldToOrigin(new Vector3(currentPos.x, 0f, currentPos.z));
            }
        }

        public void RegisterWorldRoot(Transform rootTransform)
        {
            if (rootTransform != null && !worldRootObjects.Contains(rootTransform))
                worldRootObjects.Add(rootTransform);
        }

        private void ShiftWorldToOrigin(Vector3 shiftDelta)
        {
            Physics.autoSimulation = false;

            playerBusTransform.position -= shiftDelta;

            for (int i = 0; i < worldRootObjects.Count; i++)
            {
                if (worldRootObjects[i] != null)
                    worldRootObjects[i].position -= shiftDelta;
            }

            CumulativeWorldOffset = new Vector3d(
                CumulativeWorldOffset.x + shiftDelta.x,
                CumulativeWorldOffset.y + shiftDelta.y,
                CumulativeWorldOffset.z + shiftDelta.z
            );

            Physics.SyncTransforms();
            Physics.autoSimulation = true;
        }

        public Vector3d GetTrueWorldPosition(Vector3 localUnityPosition)
        {
            return new Vector3d(
                CumulativeWorldOffset.x + localUnityPosition.x,
                CumulativeWorldOffset.y + localUnityPosition.y,
                CumulativeWorldOffset.z + localUnityPosition.z
            );
        }
    }
}

        private void Add(int id, string name, BDDivision div, string primary, string secondary,
                         RHDHighwayClass road, Vector2 coords, bool tour, bool hotelStop)
        {
            allDistricts.Add(new DistrictNode
            {
                id = id, districtName = name, division = div,
                primaryLandmark = primary, secondaryLandmarks = secondary,
                accessRoadClass = road, scaledMapCoordsMeters = coords,
                isTourModeHighlight = tour, hasMandatoryRestHotel = hotelStop
            });
        }

        public void EvaluateProximityTriggers(FloatingOriginManager.Vector3d busWorldPos, float triggerRadiusMeters = 250f)
        {
            foreach (var node in allDistricts)
            {
                double dx = busWorldPos.x - node.scaledMapCoordsMeters.x;
                double dz = busWorldPos.z - node.scaledMapCoordsMeters.y;
                double distSq = (dx * dx) + (dz * dz);

                if (distSq <= triggerRadiusMeters * triggerRadiusMeters)
                {
                    if (node.isTourModeHighlight)
                        OnTourHighlightEntered?.Invoke(node);
                    if (node.hasMandatoryRestHotel)
                        OnMandatoryHotelStopReached?.Invoke(node);
                }
            }
        }
    }

        [SerializeField] private float peakMotorTorque = 4200f;
        [SerializeField] private float maxBrakeTorque = 6500f;
        [SerializeField] private float maxSteerAngle = 34f;
        [SerializeField] private float antiRollStiffness = 18000f;
        [SerializeField] private Vector3 centerOfMassOffset = new Vector3(0f, -0.65f, 0.2f);

        [Header("Gearbox Ratios (Reverse, Neutral, 1st-6th)")]
        [SerializeField] private float[] gearRatios = { -3.8f, 0f, 4.1f, 2.5f, 1.65f, 1.18f, 0.88f, 0.72f };
        public int CurrentGearIndex { get; private set; } = 2; // 1st gear
        public float CurrentSpeedKmh { get; private set; }
        public float EngineRPM { get; private set; }

        [Header("Steering Wheel & Door Animation")]
        [SerializeField] private Transform steeringWheelMesh;
        [SerializeField] private float maxWheelVisualSpin = 450f;
        [SerializeField] private Animator passengerDoorAnimator;

        private Rigidbody rb;
        private float throttleInput;
        private float brakeInput;
        private float steerInput;
        private float currentSteerAngle;
        private bool doorsOpen;

        private void Awake()
        {
            rb = GetComponent<Rigidbody>();
            rb.mass = 13500f;
            rb.centerOfMass += centerOfMassOffset;
        }

        public void SetDrivingInputs(float throttle, float brake, float steer)
        {
            throttleInput = Mathf.Clamp01(throttle);
            brakeInput = Mathf.Clamp01(brake);
            steerInput = Mathf.Clamp(steer, -1f, 1f);
        }

        public void ShiftGearUp()
        {
            if (CurrentGearIndex < gearRatios.Length - 1)
                CurrentGearIndex++;
        }

        public void ShiftGearDown()
        {
            if (CurrentGearIndex > 0)
                CurrentGearIndex--;
        }

        public void TogglePassengerDoor()
        {
            doorsOpen = !doorsOpen;
            if (passengerDoorAnimator != null)
                passengerDoorAnimator.SetBool("IsOpen", doorsOpen);
        }

        private void FixedUpdate()
        {
            CurrentSpeedKmh = rb.velocity.magnitude * 3.6f;

            if (transmissionMode == TransmissionMode.Automatic)
                HandleAutomaticShifting();

            CalculateEngineRPM();
            ApplySteering();
            ApplyDriveAndBraking();
            ApplyAntiRollBar(frontLeftCollider, frontRightCollider);
            ApplyAntiRollBar(rearLeftCollider, rearRightCollider);

            if (CurrentSpeedKmh > 15f)
                bodyDirtiness = Mathf.Min(100f, bodyDirtiness + Time.fixedDeltaTime * 0.05f);
        }

        private void Update()
        {
            SyncWheelVisual(frontLeftCollider, frontLeftMesh);
            SyncWheelVisual(frontRightCollider, frontRightMesh);
            SyncWheelVisual(rearLeftCollider, rearLeftMesh);
            SyncWheelVisual(rearRightCollider, rearRightMesh);

            if (steeringWheelMesh != null)
            {
                float targetWheelSpin = -(currentSteerAngle / maxSteerAngle) * maxWheelVisualSpin;
                steeringWheelMesh.localRotation = Quaternion.Euler(22f, 0f, targetWheelSpin);
            }
        }

        private void HandleAutomaticShifting()
        {
            if (CurrentGearIndex <= 1) return;
            if (EngineRPM > 2250f && CurrentGearIndex < gearRatios.Length - 1)
                CurrentGearIndex++;
            else if (EngineRPM < 950f && CurrentGearIndex > 2)
                CurrentGearIndex--;
        }

        private void CalculateEngineRPM()
        {
            float wheelRPM = (rearLeftCollider.rpm + rearRightCollider.rpm) * 0.5f;
            float gearRatio = Mathf.Abs(gearRatios[CurrentGearIndex]);
            float targetRPM = 700f + Mathf.Abs(wheelRPM) * gearRatio * 2.8f + (throttleInput * 450f);
            EngineRPM = Mathf.Lerp(EngineRPM, Mathf.Clamp(targetRPM, 650f, 2600f), Time.fixedDeltaTime * 6f);
        }

        private void ApplySteering()
        {
            float speedFactor = Mathf.Lerp(1f, 0.35f, CurrentSpeedKmh / 100f);
            float targetAngle = steerInput * maxSteerAngle * speedFactor;
            currentSteerAngle = Mathf.Lerp(currentSteerAngle, targetAngle, Time.fixedDeltaTime * 8f);

            frontLeftCollider.steerAngle = currentSteerAngle;
            frontRightCollider.steerAngle = currentSteerAngle;
        }

        private void ApplyDriveAndBraking()
        {
            float healthMultiplier = Mathf.Lerp(0.45f, 1.0f, structuralHealth / 100f);
            float ratio = gearRatios[CurrentGearIndex];
            float appliedTorque = throttleInput * peakMotorTorque * (ratio / 4.1f) * healthMultiplier;

            if (CurrentGearIndex == 1) appliedTorque = 0f;

            rearLeftCollider.motorTorque = appliedTorque;
            rearRightCollider.motorTorque = appliedTorque;

            float brakeForce = brakeInput * maxBrakeTorque;
            frontLeftCollider.brakeTorque = brakeForce;
            frontRightCollider.brakeTorque = brakeForce;
            rearLeftCollider.brakeTorque = brakeForce;
            rearRightCollider.brakeTorque = brakeForce;
        }

        private void ApplyAntiRollBar(WheelCollider leftWheel, WheelCollider rightWheel)
        {
            WheelHit hit;
            float travelLeft = 1.0f;
            float travelRight = 1.0f;

            bool groundedLeft = leftWheel.GetGroundHit(out hit);
            if (groundedLeft)
                travelLeft = (-leftWheel.transform.InverseTransformPoint(hit.point).y - leftWheel.radius) / leftWheel.suspensionDistance;

            bool groundedRight = rightWheel.GetGroundHit(out hit);
            if (groundedRight)
                travelRight = (-rightWheel.transform.InverseTransformPoint(hit.point).y - rightWheel.radius) / rightWheel.suspensionDistance;

            float antiRollForce = (travelLeft - travelRight) * antiRollStiffness;

            if (groundedLeft)
                rb.AddForceAtPosition(leftWheel.transform.up * -antiRollForce, leftWheel.transform.position);
            if (groundedRight)
                rb.AddForceAtPosition(rightWheel.transform.up * antiRollForce, rightWheel.transform.position);
        }

        private void SyncWheelVisual(WheelCollider collider, Transform visualTransform)
        {
            if (collider == null || visualTransform == null) return;
            collider.GetWorldPose(out Vector3 pos, out Quaternion rot);
            visualTransform.SetPositionAndRotation(pos, rot);
        }

        private void OnCollisionEnter(Collision collision)
        {
            float impact = collision.relativeVelocity.magnitude;
            if (impact > 4f)
                structuralHealth = Mathf.Max(0f, structuralHealth - (impact * 1.5f));
        }
    }
}

using UnityEngine;

namespace BSTeam.BusSimulatorBD.Audio
{
    public class BusAudioAndHornController : MonoBehaviour
    {
        [Header("Engine Audio Clips (Extracted from Engine Shorts)")]
        [SerializeField] private AudioSource idleEngineSource;
        [SerializeField] private AudioSource midDieselSource;
        [SerializeField] private AudioSource highTurboSource;

        [Header("Horn Audio (Extracted from RbhBNRfTTTw)")]
        [SerializeField] private AudioSource hornAttackSource;
        [SerializeField] private AudioSource hornSustainLoopSource;

        [Header("RPM Pitch Calibration")]
        [SerializeField] private float baselineRPM = 1000f;
        [SerializeField] private float pitchChangePer1000RPM = 0.38f;

        private Vehicle.BDRealisticBusController busController;
        private bool isHornHeld;

        private void Awake()
        {
            busController = GetComponent<Vehicle.BDRealisticBusController>();
            ConfigureLoopingSource(idleEngineSource);
            ConfigureLoopingSource(midDieselSource);
            ConfigureLoopingSource(highTurboSource);
            ConfigureLoopingSource(hornSustainLoopSource);
        }

        private void ConfigureLoopingSource(AudioSource src)
        {
            if (src == null) return;
            src.loop = true;
            src.spatialBlend = 0.85f;
            src.playOnAwake = false;
        }

        private void Start()
        {
            idleEngineSource?.Play();
            midDieselSource?.Play();
            highTurboSource?.Play();
        }

        private void Update()
        {
            if (busController == null) return;
            float rpm = busController.EngineRPM;

            float idleWeight = Mathf.InverseLerp(1200f, 700f, rpm);
            float highWeight = Mathf.InverseLerp(1500f, 2350f, rpm);
            float midWeight = Mathf.Clamp01(1f - idleWeight - highWeight + 0.25f);

            float calculatedPitch = 1.0f + ((rpm - baselineRPM) / 1000f) * pitchChangePer1000RPM;

            if (idleEngineSource != null)
            {
                idleEngineSource.volume = idleWeight * 0.75f;
                idleEngineSource.pitch = Mathf.Clamp(calculatedPitch * 0.9f, 0.75f, 1.35f);
            }
            if (midDieselSource != null)
            {
                midDieselSource.volume = midWeight * 0.85f;
                midDieselSource.pitch = Mathf.Clamp(calculatedPitch, 0.75f, 1.45f);
            }
            if (highTurboSource != null)
            {
                highTurboSource.volume = highWeight * 0.9f;
                highTurboSource.pitch = Mathf.Clamp(calculatedPitch * 1.05f, 0.85f, 1.6f);
            }
        }

        public void OnHornButtonPressed()
        {
            if (isHornHeld) return;
            isHornHeld = true;

            if (hornAttackSource != null && hornAttackSource.clip != null)
                hornAttackSource.PlayOneShot(hornAttackSource.clip);

            if (hornSustainLoopSource != null && !hornSustainLoopSource.isPlaying)
                hornSustainLoopSource.Play();
        }

        public void OnHornButtonReleased()
        {
            isHornHeld = false;
            if (hornSustainLoopSource != null && hornSustainLoopSource.isPlaying)
                hornSustainLoopSource.Stop();
        }
    }

namespace BSTeam.BusSimulatorBD.Customization
{
    public class LiveryAndModLoader : MonoBehaviour
    {
        [Header("Target Bus Exterior Material")]
        [SerializeField] private Renderer busBodyRenderer;
        [SerializeField] private int bodyMaterialIndex = 0;
        [SerializeField] private string defaultResourcesSkinPath = "Liveries/Hanif_Enterprise_HD";

        private Texture2D activeCustomTexture;

        private void Start()
        {
            LoadBuiltInLivery(defaultResourcesSkinPath);
        }

        public void LoadBuiltInLivery(string resourcesPath)
        {
            Texture2D tex = Resources.Load<Texture2D>(resourcesPath);
            if (tex != null)
                ApplyTextureToBus(tex);
        }

        public void LoadExternalPngLivery(string absoluteFilePath)
        {
            if (string.IsNullOrEmpty(absoluteFilePath) || !File.Exists(absoluteFilePath))
            {
                Debug.LogWarning($"[LiveryLoader] Livery PNG not found: {absoluteFilePath}");
                return;
            }
            StartCoroutine(StreamPngFromStorage(absoluteFilePath));
        }

        private IEnumerator StreamPngFromStorage(string filePath)
        {
            string uri = "file://" + filePath;
            using (UnityWebRequest request = UnityWebRequestTexture.GetTexture(uri))
            {
                yield return request.SendWebRequest();

                if (request.result != UnityWebRequest.Result.Success)
                {
                    Debug.LogError($"[LiveryLoader] Failed loading PNG: {request.error}");
                    yield break;
                }

                Texture2D downloaded = DownloadHandlerTexture.GetContent(request);
                downloaded.filterMode = FilterMode.Trilinear;
                downloaded.anisoLevel = 8;

                if (activeCustomTexture != null)
                    Destroy(activeCustomTexture);

                activeCustomTexture = downloaded;
                ApplyTextureToBus(activeCustomTexture);
            }
        }

        private void ApplyTextureToBus(Texture2D skinTexture)
        {
            if (busBodyRenderer == null || skinTexture == null) return;

            Material[] mats = busBodyRenderer.materials;
            if (bodyMaterialIndex < mats.Length)
                mats[bodyMaterialIndex].SetTexture("_BaseMap", skinTexture);
        }
    }

using System.Collections.Generic;
using UnityEngine;
using GoogleMobileAds.Api;

namespace BSTeam.BusSimulatorBD.Gameplay
{
    public class DriverModeAndTrafficAdManager : MonoBehaviour
    {
        [Header("Driver Mode (Step Out at Landmarks)")]
        [SerializeField] private Transform busExitSpawnPoint;
        [SerializeField] private GameObject walkingDriverAvatar;
        [SerializeField] private Camera busFollowCamera;
        [SerializeField] private Camera driverWalkCamera;
        [SerializeField] private Vehicle.BDRealisticBusController busController;

        [Header("Bangladeshi AI Traffic Pool")]
        [SerializeField] private List<GameObject> bdTrafficPrefabs; // Local buses, trucks, CNGs
        [SerializeField] private int trafficPoolSize = 18;
        [SerializeField] private float spawnRadiusMeters = 320f;
        private readonly List<GameObject> activeTrafficPool = new List<GameObject>();

        [Header("AdMob Non-Intrusive Interstitial (Never Mid-Drive)")]
        [SerializeField] private string androidInterstitialId = "ca-app-pub-3940256099942544/1033173712"; // Google Test ID
        private InterstitialAd interstitialAd;
        private bool isDriverWalking;
        private bool isDrivingActive;

        private void Start()
        {
            InitializeTrafficPool();
            MobileAds.Initialize(initStatus => { LoadInterstitialAd(); });

            if (walkingDriverAvatar != null)
                walkingDriverAvatar.SetActive(false);
        }

        private void Update()
        {
            RecycleDistantTraffic();
        }

        public void SetDrivingSessionState(bool active)
        {
            isDrivingActive = active;
        }

        public void ToggleDriverMode()
        {
            if (busController.CurrentSpeedKmh > 2.0f) return;

            isDriverWalking = !isDriverWalking;

            if (isDriverWalking)
            {
                walkingDriverAvatar.transform.SetPositionAndRotation(
                    busExitSpawnPoint.position, busExitSpawnPoint.rotation);
                walkingDriverAvatar.SetActive(true);
                busFollowCamera.enabled = false;
                driverWalkCamera.enabled = true;
            }
            else
            {
                walkingDriverAvatar.SetActive(false);
                driverWalkCamera.enabled = false;
                busFollowCamera.enabled = true;
            }
        }

        private void InitializeTrafficPool()
        {
            if (bdTrafficPrefabs == null || bdTrafficPrefabs.Count == 0) return;

            for (int i = 0; i < trafficPoolSize; i++)
            {
                GameObject prefab = bdTrafficPrefabs[i % bdTrafficPrefabs.Count];
                GameObject vehicle = Instantiate(prefab, Vector3.zero, Quaternion.identity);
                vehicle.SetActive(false);
                activeTrafficPool.Add(vehicle);
            }
        }

        private void RecycleDistantTraffic()
        {
            if (busController == null || activeTrafficPool.Count == 0) return;
            Vector3 busPos = busController.transform.position;
            float maxDistSq = spawnRadiusMeters * spawnRadiusMeters;

            for (int i = 0; i < activeTrafficPool.Count; i++)
            {
                GameObject npc = activeTrafficPool[i];
                if (!npc.activeSelf) continue;

                if ((npc.transform.position - busPos).sqrMagnitude > maxDistSq)
                    npc.SetActive(false); // Returns AI vehicle to pool cleanly without GC
            }
        }

        public void LoadInterstitialAd()
        {
            if (interstitialAd != null)
            {
                interstitialAd.Destroy();
                interstitialAd = null;
            }

            var adRequest = new AdRequest();
            InterstitialAd.Load(androidInterstitialId, adRequest, (InterstitialAd ad, LoadAdError error) =>
            {
                if (error != null || ad == null) return;
                interstitialAd = ad;
                interstitialAd.OnAdFullScreenContentClosed += LoadInterstitialAd;
            });
        }

        public void ShowTripCompletedAd()
        {
            if (isDrivingActive) return;

            if (interstitialAd != null && interstitialAd.CanShowAd())
                interstitialAd.Show();
        }
    }
